## Python check (run with `python3 -I`)

```python
import re
def model_of(path):
    segs=[s.strip() for s in path.split('>') if s.strip()]
    ms=[s for s in segs[1:] if re.search(r'\.(nwc|rvt|ifc|dwg|nwd)$',s,re.I) and not s.lower().endswith('.nwd')]
    return ms[-1] if ms else (segs[1] if len(segs)>1 else "")
def names_match(clash,vp):
    clash=clash.strip(); vp=vp.strip()
    if not clash or not vp: return False
    if clash.lower()==vp.lower(): return True
    return re.search(r'(?<![^\W_])'+re.escape(clash)+r'(?![^\W_])',vp,re.I) is not None
```

Note: the C# uses `(?<![\p{L}\p{N}])name(?![\p{L}\p{N}])`; the Python class `[^\W_]` is the equivalent letter/digit set.
